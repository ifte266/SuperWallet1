// তোমার ফ্রন্টএন্ডের সাবমিট ফাংশনটি এভাবে রিপ্লেস করো: const submitReport = async (event) => { event.preventDefault(); const submitButton = document.querySelector('button[type="submit"]'); const statusMessage = document.getElementById('status-message') || document.createElement('div'); // বাটনে লোডিং স্টেট দেখানো if (submitButton) { submitButton.disabled = true; submitButton.innerText = 'প্রক্রিয়াধীন...'; } // ফর্ম ডাটা কালেক্ট করা const form = event.target; const formData = new FormData(form); try { // সরাসরি ভেরসেলের লাইভ ব্যাকএন্ড ইউআরএল-এ রিকোয়েস্ট পাঠানো হচ্ছে const response = await fetch('https://security-verify-g92g.vercel.app/api/submit-report', { method: 'POST', body: formData }); const result = await response.json(); if (response.ok && result.success) { statusMessage.innerText = 'রিপোর্ট সফলভাবে পাঠানো হয়েছে।'; statusMessage.style.color = '#10B981'; // সবুজ রঙ form.reset(); // ফর্ম খালি করা } else { statusMessage.innerText = result.error || 'টেলিগ্রামে সেন্ড করতে ব্যর্থ।'; statusMessage.style.color = '#EF4444'; // লাল রঙ } } catch (error) { console.error('Fetch Error:', error); statusMessage.innerText = 'সার্ভারের সাথে যোগাযোগ করা যাচ্ছে না।'; statusMessage.style.color = '#EF4444'; // লাল রঙ } finally { if (submitButton) { submitButton.disabled = false; submitButton.innerText = 'নিরাপদে রিপোর্ট সাবমিট করুন'; } } };
